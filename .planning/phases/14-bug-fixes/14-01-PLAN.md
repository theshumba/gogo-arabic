    **Step 2: Create `useOverlayClose` hook**

    Create new file `src/hooks/useOverlayClose.js`. This centralizes ESC key, click-outside, and guaranteed unfreeze:

    ```js
    import { useEffect, useCallback } from 'react';
    import { EventBus } from '../utils/eventBus.js';

    /**
     * useOverlayClose — centralized overlay dismissal
     * Handles: ESC key, optional click-outside, guaranteed unfreeze-player
     *
     * @param {function} onClose - The close handler (dispatch closeX action, etc.)
     * @param {object} options
     * @param {boolean} options.escEnabled - Allow ESC to close (default: true)
     * @param {function} options.beforeClose - Optional guard (return false to prevent close)
     */
    export function useOverlayClose(onClose, options = {}) {
      const { escEnabled = true, beforeClose = null } = options;

      const handleClose = useCallback(() => {
        if (beforeClose && beforeClose() === false) return;
        onClose();
        EventBus.emit('unfreeze-player');
      }, [onClose, beforeClose]);

      // ESC key handler
      useEffect(() => {
        if (!escEnabled) return;

        const handleKeyDown = (e) => {
          if (e.key === 'Escape') {
            e.preventDefault();
            e.stopPropagation();
            handleClose();
          }
        };
        window.addEventListener('keydown', handleKeyDown, true); // capture phase
        return () => window.removeEventListener('keydown', handleKeyDown, true);
      }, [escEnabled, handleClose]);

      // Unmount safety net — if overlay is removed without explicit close, still unfreeze
      useEffect(() => {
        return () => {
          EventBus.emit('unfreeze-player');
        };
      }, []);

      return handleClose;
    }
    ```

    IMPORTANT: The unmount cleanup `useEffect(() => { return () => EventBus.emit('unfreeze-player'); }, [])` is the critical safety net. If any overlay unmounts for ANY reason (route change, error boundary, parent re-render), the player always gets unfrozen. This is idempotent — calling unfreeze when already unfrozen is a no-op (`isFrozen = false` is already false).

    **Part A: Zone Transition Freeze Fix**

    Root causes from code review:
    1. `ZoneTransition.transitionTo()` is async and uses `await new Promise(resolve => camera.once('camerafadeoutcomplete', resolve))`. If the camera event NEVER fires (scene destroyed during transition, camera replaced during loadZone, etc.), the promise hangs forever, `transitioning` stays `true`, and `unfreeze-player` never emits.
    2. `loadZone()` in WorldScene calls `clearZone()` which destroys the player, then `buildZone()` creates a new one. If the camera fade events are still pending from the old scene state, they may reference destroyed objects.
    3. No timeout protection — if transition hangs, it hangs forever.

    **Fix ZoneTransition.js (`src/game/systems/ZoneTransition.js`):**

    Wrap the entire transition in try/catch/finally with timeout protection:

    ```js
    import { EventBus } from '../../utils/eventBus.js';

    class ZoneTransition {
      constructor(scene) {
        this.scene = scene;
        this.transitioning = false;
      }

      async transitionTo(zoneName, entryX, entryY) {
        if (this.transitioning) return;
        this.transitioning = true;

        // Disable player movement during transition
        EventBus.emit('freeze-player');

        try {
          // Fade out with timeout protection (3 second max)
          await this._fadeWithTimeout('fadeOut', 500, 3000);

          // Notify Redux of zone change
          EventBus.emit('zone-change', { zone: zoneName, x: entryX, y: entryY });

          // Load the new zone
          this.scene.loadZone(zoneName, entryX, entryY);

          // Fade in with timeout protection
          await this._fadeWithTimeout('fadeIn', 500, 3000);
        } catch (err) {
          console.error('Zone transition error:', err);
          // Force camera to normal state on error
          try {
            this.scene.cameras.main.resetFX();
          } catch (_) {
            // Camera may not exist if scene was destroyed
          }
        } finally {
          // ALWAYS reset state and unfreeze, even on error
          this.transitioning = false;
          EventBus.emit('unfreeze-player');
        }
      }

      /**
       * Camera fade with timeout protection.
       * Resolves when fade completes OR when timeout expires (whichever first).
       */
      _fadeWithTimeout(fadeType, duration, timeout) {
        return new Promise((resolve) => {
          const camera = this.scene?.cameras?.main;
          if (!camera) {
            resolve();
            return;
          }

          let settled = false;
          const settle = () => {
            if (settled) return;
            settled = true;
            clearTimeout(timer);
            resolve();
          };

          // Start the fade
          if (fadeType === 'fadeOut') {
            camera.fadeOut(duration, 0, 0, 0);
            camera.once('camerafadeoutcomplete', settle);
          } else {
            camera.fadeIn(duration, 0, 0, 0);
            camera.once('camerafadeincomplete', settle);
          }

          // Timeout safety — resolve even if camera event never fires
          const timer = setTimeout(settle, timeout);
        });
      }
    }

    export default ZoneTransition;
    ```
