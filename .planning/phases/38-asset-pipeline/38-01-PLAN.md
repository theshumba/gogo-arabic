```bash
const fs = require('fs');
const path = require('path');

const SRC_BASE = '/tmp/kenmi';
const DST_BASE = '/Users/theshumba/Documents/GitHub/gogo-arabic/public/assets/kenmi';

const packMap = {
  'Cute_Fantasy/Cute_Fantasy': 'base',
  'Cute_Fantasy_Desert/Cute_Fantasy_Desert': 'desert',
  'Cute_Fantasy_Dungeons/Cute_Fantasy_Dungeons': 'dungeons',
  'Cute_Fantasy_UI/Cute_Fantasy_UI': 'ui',
  'Cute_Fantasy_Characters/Cute_Fantasy_Characters': 'characters',
  'Cute_Fantasy_ShroomLands/Cute_Fantasy_ShroomLands': 'shroomlands',
  'Cute_Fantasy_MilitaryCamp/Cute_Fantasy_MilitaryCamp': 'militarycamp',
  'Cute_Fantasy_Halloween/Cute_Fantasy_Halloween': 'halloween',
  'Cute_Fantasy_Christmas/Cute_Fantasy_Christmass': 'christmas',
  'Cute_Fantasy_Volcano/Cute_Fantasy_Volcano': 'volcano',
};

function toKebab(str) {
  // Preserve file extension dots, convert separators
  return str
    .replace(/\s+/g, '-')
    .replace(/_/g, '-')
    .toLowerCase();
}

function walkAndCopy(srcDir, dstDir) {
  fs.mkdirSync(dstDir, { recursive: true });
  for (const entry of fs.readdirSync(srcDir, { withFileTypes: true })) {
    const kebabName = toKebab(entry.name);
    const srcPath = path.join(srcDir, entry.name);
    const dstPath = path.join(dstDir, kebabName);
    if (entry.isDirectory()) {
      walkAndCopy(srcPath, dstPath);
    } else if (entry.isFile() && entry.name.endsWith('.png')) {
      fs.copyFileSync(srcPath, dstPath);
    }
  }
}

for (const [srcRelative, packName] of Object.entries(packMap)) {
  const srcDir = path.join(SRC_BASE, srcRelative);
  const dstDir = path.join(DST_BASE, packName);
  console.log('Copying', packName, '...');
  walkAndCopy(srcDir, dstDir);
  console.log('Done:', packName);
}
console.log('All packs copied.');
"
```
