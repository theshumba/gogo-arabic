  ```javascript
  import { EventBus } from '../../../utils/eventBus.js';
  import { EVENTS } from '../../../utils/eventBusTypes.js';
  import { store } from '../../../store/store.js';
  import { startArenaChallenge, completeArenaWave, endArenaChallenge, submitArenaScore } from '../../../store/slices/arenaSlice.js';
  import { getWaveConfig, ARENA_MODES } from '../../../data/arenaChallenges.js';
  import { getEnemy } from '../../../data/enemies.js';

  export class ArenaController {
    constructor(scene, arenaConfig) {
      this.scene = scene;
      this.config = arenaConfig; // { mode: 'survival'|'boss_rush'|'puzzle', ... }
      this.currentWave = 0;
      this.score = 0;
      this.streak = 0;
      this.waveStartTime = null;
      this.totalAccuracy = 0;
      this.totalAnswers = 0;
      this.isActive = false;
    }

    start() — dispatches startArenaChallenge, calls startWave()

    startWave() {
      this.currentWave++;
      this.waveStartTime = Date.now();

      const waveConfig = getWaveConfig(this.currentWave);

      // Generate enemy party for this wave
      const enemyParty = this._generateWaveEnemies(waveConfig);

      // Emit wave start event (React ArenaHUD listens)
      EventBus.emit(EVENTS.ARENA_WAVE_START, {
        wave: this.currentWave,
        enemies: enemyParty,
        arabicDifficulty: waveConfig.arabicDifficulty,
        timeLimit: waveConfig.timeLimit,
        bonusObjective: waveConfig.bonusObjective,
      });

      return { enemyParty, waveConfig };
    }

    onWaveComplete(battleResult) {
      const waveTime = Date.now() - this.waveStartTime;
      const waveScore = this._calculateWaveScore(battleResult, waveTime);
      this.score += waveScore;

      // Track accuracy
      if (battleResult.accuracy !== undefined) {
        this.totalAccuracy += battleResult.accuracy;
        this.totalAnswers++;
      }

      if (battleResult.victory) {
        this.streak++;

        // Dispatch wave completion to Redux
        store.dispatch(completeArenaWave({
          score: waveScore,
          accuracy: battleResult.accuracy || 0,
          enemiesDefeated: battleResult.enemiesDefeated || 1,
        }));

        EventBus.emit(EVENTS.ARENA_WAVE_COMPLETE, {
          wave: this.currentWave,
          score: waveScore,
          totalScore: this.score,
          streak: this.streak,
        });

        // Check arena completion
        const maxWaves = this.config.maxWaves || ARENA_MODES[this.config.mode]?.maxWaves || 10;
        if (this.currentWave >= maxWaves) {
          this._completeArena();
        } else {
          // Brief pause between waves (3 seconds)
          // Clean up sprite pools between waves
          this._cleanupBetweenWaves();
          this.scene.time.delayedCall(3000, () => this.startWave());
        }
      } else {
        this._failArena();
      }
    }

    _calculateWaveScore(result, waveTime) {
      const baseScore = 100;
      const accuracyBonus = (result.accuracy || 0) * 50;  // Up to 50 points
      const speedBonus = Math.max(0, 50 - Math.floor(waveTime / 1000)); // Up to 50 points
      const streakBonus = this.streak * 20;  // 20 per consecutive wave
      const bonusObjectivePoints = result.bonusCompleted ? 100 : 0;

      return Math.floor(baseScore + accuracyBonus + speedBonus + streakBonus + bonusObjectivePoints);
    }

    }

    _cleanupBetweenWaves() {

      // this.scene.sprites.clearEnemies()
      // this.scene.damagePool.clear()
      // this.scene.effects.clearActive()
      // Do NOT recreate scene — reuse same instance
    }

    _completeArena() {
      this.isActive = false;
      const avgAccuracy = this.totalAnswers > 0 ? this.totalAccuracy / this.totalAnswers : 0;

      store.dispatch(endArenaChallenge({ victory: true, finalScore: this.score }));
      store.dispatch(submitArenaScore({
        mode: this.config.mode,
        score: this.score,
        wavesCompleted: this.currentWave,
        accuracy: avgAccuracy,
        timestamp: Date.now(),
      }));

      EventBus.emit(EVENTS.ARENA_COMPLETE, {
        victory: true,
        score: this.score,
        wavesCompleted: this.currentWave,
        accuracy: avgAccuracy,
        streak: this.streak,
      });
    }

    _failArena() {
      this.isActive = false;
      const avgAccuracy = this.totalAnswers > 0 ? this.totalAccuracy / this.totalAnswers : 0;

      store.dispatch(endArenaChallenge({ victory: false, finalScore: this.score }));
      store.dispatch(submitArenaScore({
        mode: this.config.mode,
        score: this.score,
        wavesCompleted: this.currentWave - 1,
        accuracy: avgAccuracy,
        timestamp: Date.now(),
      }));

      EventBus.emit(EVENTS.ARENA_COMPLETE, {
        victory: false,
        score: this.score,
        wavesCompleted: this.currentWave - 1,
        accuracy: avgAccuracy,
      });
    }

    destroy() {
      this.isActive = false;
      this._cleanupBetweenWaves();
    }
  }
  ```

  Key design decisions:
  - Accuracy rewarded MORE than speed (50 accuracy points max vs 50 speed points max)
  - Streak bonus incentivizes consistency
  - Memory cleanup between waves is critical (sprites, damage pool, effects)
  - 3-second pause between waves for player breathing room
