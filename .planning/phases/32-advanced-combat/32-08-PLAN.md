  export class BossRushController {
    constructor(scene) {
      this.scene = scene;
      this.currentBossIndex = 0;
      this.totalScore = 0;
      this.bossResults = []; // [{ bossId, victory, score, accuracy, time }]
      this.isActive = false;
    }

    /**
     * Check if boss rush is unlocked.
     * Requires all bosses in BOSS_RUSH_SEQUENCE defeated in story mode.
     */
    static isUnlocked() {
      const defeated = store.getState().battle.bossesDefeated;
      return BOSS_RUSH_SEQUENCE.every(entry => defeated.includes(entry.bossId));
    }

    start() — verify unlocked, set isActive, startNextBoss()

    startNextBoss() {
      const entry = BOSS_RUSH_SEQUENCE[this.currentBossIndex];
      if (!entry) {
        this._completeRush();
        return;
      }

      // Show interlude first (if not first boss)
      if (this.currentBossIndex > 0) {
        const prevEntry = BOSS_RUSH_SEQUENCE[this.currentBossIndex - 1];
        EventBus.emit(EVENTS.BATTLE_STATE_CHANGED, {
          to: 'BOSS_RUSH_INTERLUDE',
          interlude: prevEntry.interlude,
          nextBoss: entry.bossId,
        });
        // Wait for interlude dismissal (React component emits continue)
      } else {
        this._launchBoss(entry);
      }
    }

    onInterludeDismissed() {
      const entry = BOSS_RUSH_SEQUENCE[this.currentBossIndex];
      this._launchBoss(entry);
    }

    _launchBoss(entry) {
      // Configure battle for this boss
      // Emit event to start battle with boss config
    }

    onBossDefeated(result) {
      this.bossResults.push({
        bossId: BOSS_RUSH_SEQUENCE[this.currentBossIndex].bossId,
        victory: result.victory,
        score: result.score || 0,
        accuracy: result.accuracy || 0,
        time: result.timeElapsed || 0,
      });

      this.totalScore += result.score || 0;
      this.currentBossIndex++;

      if (!result.victory) {
        // Boss rush failed on this boss
        this._failRush();
        return;
      }

      // Continue to next boss
      this.startNextBoss();
    }

    _completeRush() {
      this.isActive = false;
      // Submit total score to arena leaderboard
      store.dispatch(submitArenaScore({
        mode: 'boss_rush',
        score: this.totalScore,
        wavesCompleted: this.bossResults.length,
        accuracy: avg(this.bossResults.map(r => r.accuracy)),
        timestamp: Date.now(),
      }));
      EventBus.emit(EVENTS.ARENA_COMPLETE, { ... });
    }

    _failRush() {
      this.isActive = false;
      store.dispatch(submitArenaScore({ ... }));
      EventBus.emit(EVENTS.ARENA_COMPLETE, { victory: false, ... });
    }

    destroy() { this.isActive = false; }
  }
