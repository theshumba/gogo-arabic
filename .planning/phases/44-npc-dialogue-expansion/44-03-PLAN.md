1. **Validates all teachWord references**: Extract every teachWord value from npcs.json. Cross-reference against vocabulary.json and vocabulary-final.json. Report any invalid IDs (teachWord values that don't match any vocabulary word ID).

2. **Counts total dialogue lines**: Count all lines across all NPCs that have a "speaker" field or an "arabic" field (excluding quiz action blocks). Verify the total is 1,543+.

3. **Counts cultural notes**: Total culturalNote fields across all NPCs. Should be 48+ (24 NPCs x 2 minimum).

4. **Validates personality distinctness**: For each of the 23 main NPCs, extract the first line of their intro tree. Verify no two NPCs share identical opening lines.

5. **Historical fact check**: Count NPCs with culturalNote fields that reference specific dates, people, or places. Should be 8+ (DIAL-02 requirement).

Create and run a Node.js verification script:

```bash
node -e "
const npcs = require('./src/data/npcs.json');
const vocab = require('./src/data/vocabulary.json');
const vocabFinal = require('./src/data/vocabulary-final.json');
const allIds = new Set([...vocab.map(w=>w.id), ...vocabFinal.map(w=>w.id)]);

let totalLines = 0, totalTeachWords = 0, totalCulturalNotes = 0;
let invalidTeachWords = [];
let historicalNpcs = new Set();

npcs.forEach(npc => {
  npc.dialogueTrees?.forEach(tree => {
    tree.lines?.forEach(line => {
      if (line.speaker || line.arabic) totalLines++;
      if (line.teachWord) {
        totalTeachWords++;
        if (!allIds.has(line.teachWord)) invalidTeachWords.push({npc: npc.id, word: line.teachWord});
      }
      if (line.culturalNote) {
        totalCulturalNotes++;
        if (/\\d{3,4}\\s*(CE|BCE|century)/i.test(line.culturalNote)) historicalNpcs.add(npc.id);
      }
    });
  });
});

console.log('Total lines:', totalLines, totalLines >= 1543 ? 'PASS' : 'FAIL');
console.log('Total teachWords:', totalTeachWords);
console.log('Total culturalNotes:', totalCulturalNotes, totalCulturalNotes >= 48 ? 'PASS' : 'FAIL');
console.log('Invalid teachWord refs:', invalidTeachWords.length, invalidTeachWords.length === 0 ? 'PASS' : 'FAIL');
if (invalidTeachWords.length > 0) console.log('  Invalid:', JSON.stringify(invalidTeachWords));
console.log('NPCs with historical facts:', historicalNpcs.size, historicalNpcs.size >= 8 ? 'PASS' : 'FAIL');
"
```
