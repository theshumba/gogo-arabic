    ```js
    #!/usr/bin/env node
    /**
     * Capture deterministic world snapshots for all 8 core zones + 3 interior scenes.
     *
     * Usage:  npm run capture:world-snapshots
     * Output: src/test/fixtures/world-snapshots/{zoneId}.json (11 files)
     *
     * Determinism contract:
     *   - MapLoader runs against a headless scene mock (sceneMock.js)
     *   - Frame totals seeded from KENMI_FRAME_TABLES to mirror BootScene preload
     *   - captureZoneSnapshot serializes in (y * mapW + x) order
     */

    import fs from 'node:fs';
    import path from 'node:path';

    // Dynamic imports so this runs as ESM in Node without TS/Vite tooling
    const { createMockScene } = await import('../src/game/systems/__tests__/mocks/sceneMock.js');
    const { MapLoader } = await import('../src/game/systems/MapLoader.js');
    const { captureZoneSnapshot } = await import('../src/game/systems/world/WorldSnapshot.js');
    const { ZONES } = await import('../src/data/zones.js');
    const { KENMI_FRAME_TABLES } = await import('../src/data/kenmiFrameTables.js');

    // Try to load interiors if the module exists
    let INTERIORS = {};
    try {
      const mod = await import('../src/data/interiors.js');
      INTERIORS = mod.INTERIORS || mod.default || {};
    } catch (e) {
      console.warn('[capture] interiors.js not importable; skipping interior snapshots:', e.message);
    }

    const CORE_ZONES = ['oasis_village', 'ancient_library', 'desert_marketplace', 'farmland', 'bedouin_camp', 'mountain_village', 'coastal_port', 'royal_palace'];
    const CORE_INTERIORS = ['scholar_house_interior', 'merchant_house_interior', 'oasis_guild_interior'];

    const OUT_DIR = path.resolve('src/test/fixtures/world-snapshots');
    fs.mkdirSync(OUT_DIR, { recursive: true });

    function buildScene() {
      const scene = createMockScene();
      // Seed every KENMI_FRAME_TABLES entry into scene.textures so _safeFrame, frameValidity,
      // and snapshot key/frame lookups match runtime behavior.
      for (const [key, table] of Object.entries(KENMI_FRAME_TABLES)) {
        scene.textures._setFrameTotal(key, table.totalFrames);
      }
      return scene;
    }

    function captureAndWrite(zoneId, zone) {
      const scene = buildScene();
      const loader = new MapLoader(scene);
      loader.create(zone, zone.mapWidth, zone.mapHeight);
      const snapshot = captureZoneSnapshot(loader, zone, zone.tilesetTheme);
      const outPath = path.join(OUT_DIR, `${zoneId}.json`);
      fs.writeFileSync(outPath, JSON.stringify(snapshot, null, 2) + '\n');
      console.log(`wrote ${outPath} — ${snapshot.tiles.length} tiles, ${snapshot.objects.length} objects, ${snapshot.decoCount} deco, ${snapshot.animalCount} animals`);
    }

    for (const zoneId of CORE_ZONES) {
      const zone = ZONES[zoneId];
      if (!zone) {
        console.error(`[capture] ZONES['${zoneId}'] not found — skipping`);
        continue;
      }
      captureAndWrite(zoneId, zone);
    }

    for (const interiorId of CORE_INTERIORS) {
      const interior = INTERIORS[interiorId];
      if (!interior) {
        console.warn(`[capture] INTERIORS['${interiorId}'] not found — skipping interior`);
        continue;
      }
      captureAndWrite(interiorId, interior);
    }

    console.log('\nDone. Review fixture files, then commit.');
    ```
