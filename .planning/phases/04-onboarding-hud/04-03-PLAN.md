setOnboardingHighlight(visible) {
  if (this._onboardingVisible === visible) return; // Skip if no change
  this._onboardingVisible = visible;

  this.onboardingArrow.setVisible(visible);
  this.onboardingGlow.setVisible(visible);

  if (visible) {

      y: this.y - 110,
      duration: 600,

    });
  } else {
    this.scene.tweens.killTweensOf(this.onboardingArrow);

**Create ContextualOnboarding.jsx:**

```jsx
import { useState, useEffect, useCallback } from 'react';
import Joyride, { STATUS } from 'react-joyride';
import { useSelector, useDispatch } from 'react-redux';
import { completeOnboarding, setOnboardingStep, setOnboardingTargetNpc } from '../../store/slices/playerSlice.js';
import { EventBus } from '../../utils/eventBus.js';
import { onboardingSteps } from './onboardingSteps.js';

// Scholar Yusuf's NPC ID from src/data/zones.js (oasis_village zone, id: 'scholar-yusuf')
// and src/data/npcs.json (id: 'scholar-yusuf'). Matches npc.npcId in Phaser.
const SCHOLAR_NPC_ID = 'scholar-yusuf';

export default function ContextualOnboarding() {
  const dispatch = useDispatch();
  const onboardingComplete = useSelector((state) => state.player.onboardingComplete);
  const onboardingStep = useSelector((state) => state.player.onboardingStep ?? 0);

  const [stepIndex, setStepIndex] = useState(onboardingStep);
  const [run, setRun] = useState(!onboardingComplete);

  // Don't render if onboarding is already complete
  if (onboardingComplete) return null;

  // Set Scholar Yusuf highlight when reaching the final step
  useEffect(() => {
    if (stepIndex >= onboardingSteps.length - 1) {
      dispatch(setOnboardingTargetNpc(SCHOLAR_NPC_ID));
    }
    return () => {
      // Clean up highlight on unmount
      dispatch(setOnboardingTargetNpc(null));
    };
  }, [stepIndex, dispatch]);

  // Listen for EventBus trigger to advance from step 0 to step 1
  useEffect(() => {
    if (stepIndex !== 0) return; // Only listen when on step 0

    const currentStep = onboardingSteps[1]; // Next step's trigger
    if (!currentStep?.trigger) return;

    let triggered = false;
    const handler = () => {
      if (triggered) return;
      triggered = true;
      // Player moved — advance to step 1
      setStepIndex(1);
      dispatch(setOnboardingStep(1));
    };

    EventBus.on(currentStep.trigger, handler);
    return () => {
      EventBus.off(currentStep.trigger, handler);
    };
  }, [stepIndex, dispatch]);

  const handleJoyrideCallback = useCallback((data) => {
    const { status, index, action, type } = data;

    // Track step changes
    if (type === 'step:after') {
      const nextIndex = index + 1;

      // Check if next step has a trigger — if so, don't auto-advance
      // (The EventBus listener will advance when trigger fires)
      const nextStep = onboardingSteps[nextIndex];
      if (nextStep?.trigger && nextIndex > stepIndex) {
        // Pause at current position and wait for trigger
        setRun(false);
        return;
      }

      if (nextIndex < onboardingSteps.length) {
        setStepIndex(nextIndex);
        dispatch(setOnboardingStep(nextIndex));
      }
    }

    // Handle completion or skip
    if (status === STATUS.FINISHED || status === STATUS.SKIPPED) {
      dispatch(completeOnboarding());
      dispatch(setOnboardingTargetNpc(null));
      dispatch(setOnboardingStep(onboardingSteps.length));
    }
  }, [dispatch, stepIndex]);

  // Resume Joyride when stepIndex advances (from EventBus trigger)
  useEffect(() => {
    if (stepIndex > 0 && !run) {
      setRun(true);
    }
  }, [stepIndex]);

  return (
    <Joyride
      steps={onboardingSteps}
      run={run}
      stepIndex={stepIndex}
      continuous
      showSkipButton
      disableOverlayClose
      hideCloseButton={false}
      callback={handleJoyrideCallback}
      locale={{
        back: 'Back',
        close: 'Close',
        last: 'Start Playing!',
        next: 'Next',
        skip: 'Skip Tutorial',
      }}
      styles={{
        options: {
          primaryColor: '#e2b659',
          textColor: '#2b292c',
          backgroundColor: '#f5f3da',
          arrowColor: '#f5f3da',
          zIndex: 10000,
        },
        tooltip: {
          fontFamily: "'Press Start 2P', cursive",
          fontSize: '11px',
          lineHeight: '1.8',
          padding: '20px',
          borderRadius: '0px',
          border: '4px solid #2b292c',
          boxShadow: 'inset -4px -4px 0px 0px rgba(0,0,0,0.2), inset 4px 4px 0px 0px rgba(255,255,255,0.3), 0 8px 0 0 rgba(0,0,0,0.3)',
        },
        buttonNext: {
          fontFamily: "'Press Start 2P', cursive",
          fontSize: '10px',
          backgroundColor: '#e2b659',
          color: '#2b292c',
          borderRadius: '0px',
          padding: '10px 20px',
          border: 'none',
        },
        buttonBack: {
          fontFamily: "'Press Start 2P', cursive",
          fontSize: '10px',
          color: '#3a373b',
        },
        buttonSkip: {
          fontFamily: "'Press Start 2P', cursive",
          fontSize: '9px',
          color: '#3a373b',
        },
        spotlight: {
          borderRadius: '0px',
        },
      }}
    />
  );
}
```
