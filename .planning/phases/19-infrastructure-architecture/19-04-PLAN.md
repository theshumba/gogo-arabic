Follow the WorldScene delegation pattern exactly (`this.systemName = new SystemClass(this)` in create, `destroy()` in clearZone/shutdown). Import `{ EVENTS }` from `../../utils/eventBusTypes.js` for any EventBus calls.

```javascript
export class SceneStackManager {
  constructor(scene) {
    this.scene = scene;  // WorldScene reference
    this._stack = [];     // Track pushed scene keys for pop behavior
  }

  /**
   * Enter a building — pause WorldScene, launch interior scene.
   * Respects prefers-reduced-motion for any transition effects (INFR-10).
   */
  pushScene(interiorSceneKey, data = {}) {
    this._stack.push(this.scene.scene.key);
    this.scene.scene.pause();
    this.scene.scene.launch(interiorSceneKey, {
      ...data,
      returnSceneKey: this.scene.scene.key,
    });
  }

  /**
   * Exit a building — stop interior scene, resume WorldScene.
   * Interior scene.stop() triggers its shutdown() for cleanup.
   */
  popScene() {
    if (this._stack.length === 0) return null;

    // Find the currently active non-WorldScene scene
    const activeScenes = this.scene.sys.scene.manager.getActiveScenes();
    const interiorScene = activeScenes.find(s => s !== this.scene && s.scene.key !== this.scene.scene.key);

    if (interiorScene) {
      interiorScene.scene.stop();  // stop() triggers shutdown() for cleanup
    }

    this.scene.scene.resume();
    return this._stack.pop();
  }

  /** Check if we're inside a building (stack depth > 0) */
  get isInBuilding() {
    return this._stack.length > 0;
  }

  /** Get current stack depth */
  get depth() {
    return this._stack.length;
  }

  destroy() {
    // Pop all remaining scenes to prevent orphans
    while (this._stack.length > 0) {
      this.popScene();
    }
    this._stack = [];
  }
}
```

{

  buildStart() {
    // Only validate during build (not dev server start)
