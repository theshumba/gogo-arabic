    ### 2. Create WelcomeBackOverlay.jsx

    ```jsx
    // Imports
    import { useSelector, useDispatch } from 'react-redux';
    import { motion } from 'framer-motion';
    import { selectLastSessionSummary, selectWelcomeBackShown, markWelcomeBackShown } from '../../store/slices/dailyGoalsSlice.js';
    import { selectFsrsCards } from '../../store/slices/vocabularySlice.js';
    import { getDueCards } from '../../services/fsrs.js';
    import { selectLearnedWordCount } from '../../store/slices/vocabularySlice.js';
    import styles from './WelcomeBackOverlay.module.css';

    // Props: { onDismiss, onNavigate }
    ```

    **Component Logic:**

    a. Read Redux state:
    ```js
    const dispatch = useDispatch();
    const lastSession = useSelector(selectLastSessionSummary);
    const fsrsCards = useSelector(selectFsrsCards);
    const playerState = useSelector(state => state.player);
    const learnedWordCount = useSelector(selectLearnedWordCount);
    ```

    b. Compute display data:
    ```js
    const dueCards = getDueCards(fsrsCards);
    const dueCount = dueCards.length;

    const streak = playerState.streak || 0;
    const lastPlayedDate = playerState.lastPlayedDate;

    // Determine streak status
    const today = new Date().toDateString();
    const yesterday = new Date(Date.now() - 86400000).toDateString();
    const streakMaintained = lastPlayedDate === today || lastPlayedDate === yesterday;
    const streakBroken = !streakMaintained && streak > 0;
    ```

    c. Determine suggested activity (mirror DailyDashboard logic):
    ```js
    let suggestedActivity, suggestedLabel, suggestedIcon;
    if (dueCount > 0) {
      suggestedActivity = 'review';
      suggestedLabel = `Review ${dueCount} due word${dueCount !== 1 ? 's' : ''}`;
      suggestedIcon = '📖';
    } else if (learnedWordCount < 10) {
      suggestedActivity = 'alphabet';
      suggestedLabel = 'Continue Learning Letters';
      suggestedIcon = '✏️';
    } else {
      suggestedActivity = 'explore';
      suggestedLabel = 'Explore the World';
      suggestedIcon = '🗺️';
    }
    ```

    d. Handle dismiss:
    ```js
    const handleDismiss = useCallback(() => {
      dispatch(markWelcomeBackShown());
      onDismiss();
    }, [dispatch, onDismiss]);

    const handleActivity = useCallback(() => {
      dispatch(markWelcomeBackShown());
      onNavigate(suggestedActivity);
    }, [dispatch, onNavigate, suggestedActivity]);
    ```

    e. Render structure:
    ```jsx
    <div className={styles.backdrop} onClick={handleDismiss}>
      <motion.div
        className={styles.panel}
        onClick={e => e.stopPropagation()}
        initial={{ opacity: 0, scale: 0.9, y: 20 }}
        animate={{ opacity: 1, scale: 1, y: 0 }}
        transition={{ duration: 0.3 }}
      >
        {/* Header */}
        <h2 className={styles.title}>مَرحَباً بِعَودَتِك!</h2>
        <p className={styles.subtitle}>Welcome Back!</p>

        {/* Last Session Summary */}
        <div className={styles.summarySection}>
          <h3 className={styles.sectionTitle}>Last Session</h3>
          <div className={styles.statRow}>
            <div className={styles.stat}>
              <span className={styles.statValue}>{lastSession.wordsLearned}</span>
              <span className={styles.statLabel}>Words Learned</span>
            </div>
            <div className={styles.stat}>
              <span className={styles.statValue}>{lastSession.reviewsDone}</span>
              <span className={styles.statLabel}>Reviews Done</span>
            </div>
          </div>
        </div>

        {/* FSRS Due Cards */}
        {dueCount > 0 && (
          <div className={styles.dueSection}>
            <span className={styles.dueCount}>{dueCount}</span>
            <span className={styles.dueLabel}>
              word{dueCount !== 1 ? 's' : ''} ready for review
            </span>
          </div>
        )}

        {/* Streak Status */}
        <div className={styles.streakSection}>
          {streakBroken ? (
            <p className={styles.streakBroken}>
              Your {streak}-day streak ended. Start a new one today!
            </p>
          ) : streak > 0 ? (
            <p className={styles.streakActive}>
              🔥 {streak}-day streak! Keep it going!
            </p>
          ) : (
            <p className={styles.streakNew}>
              Start your first streak today!
            </p>
          )}
        </div>

        {/* CTA Buttons */}
        <button className={styles.ctaButton} onClick={handleActivity}>
          {suggestedLabel}
        </button>
        <button className={styles.skipButton} onClick={handleDismiss}>
          Skip
        </button>
      </motion.div>
    </div>
    ```

    ### 4. Wire into GameLayout.jsx

    a. Add lazy import (at top with other lazy imports):
    ```js
    const WelcomeBackOverlay = lazy(() => import('../WelcomeBack/WelcomeBackOverlay.jsx'));
    ```

    b. Add state:
    ```js
    const [showWelcomeBack, setShowWelcomeBack] = useState(false);
    ```

    c. Add selectors:
    ```js
    const welcomeBackShown = useSelector(selectWelcomeBackShown);
    const lastPlayedDate = useSelector(state => state.player?.lastPlayedDate);
    ```

    d. Add useEffect to check on mount whether to show:
    ```js
    useEffect(() => {
      if (welcomeBackShown) return; // Already shown this session

      if (!lastPlayedDate) return; // First-time player — no welcome back

      const lastPlayed = new Date(lastPlayedDate);
      const hoursSince = (Date.now() - lastPlayed.getTime()) / (1000 * 60 * 60);

      if (hoursSince >= 4) {
        setShowWelcomeBack(true);
      }
    }, []); // Only on mount
    ```

    e. Add handlers:
    ```js
    const handleWelcomeBackDismiss = useCallback(() => {
      setShowWelcomeBack(false);
      EventBus.emit(EVENTS.PLAYER_UNFREEZE);
    }, []);

    const handleWelcomeBackNavigate = useCallback((activity) => {
      setShowWelcomeBack(false);
      // Route to the suggested activity
      switch (activity) {
        case 'review':
          // Open quiz overlay or daily dashboard for review
          // Use existing navigation pattern from GameLayout
          break;
        case 'alphabet':
          // Navigate to alphabet learning
          break;
        case 'explore':
        default:
          // Just close overlay — player is already in the world
          break;
      }
      EventBus.emit(EVENTS.PLAYER_UNFREEZE);
    }, []);
    ```

    f. When showWelcomeBack is true, freeze player:
    ```js
    useEffect(() => {
      if (showWelcomeBack) {
        EventBus.emit(EVENTS.PLAYER_FREEZE);
      }
    }, [showWelcomeBack]);
    ```

    g. Add overlay render (after other overlays):
    ```jsx
    {showWelcomeBack && (
      <Suspense fallback={null}>
        <WelcomeBackOverlay
          onDismiss={handleWelcomeBackDismiss}
          onNavigate={handleWelcomeBackNavigate}
        />
      </Suspense>
    )}
    ```
