```js
adjustFriendship(state, action) {
  // payload: { npcId, delta, reason }
  const { npcId, delta } = action.payload;
  if (!state.friendship[npcId]) state.friendship[npcId] = 50; // Start neutral
  state.friendship[npcId] = Math.max(0, Math.min(100, state.friendship[npcId] + delta));
},

setFriendship(state, action) {
  // payload: { npcId, value }
  const { npcId, value } = action.payload;
  state.friendship[npcId] = Math.max(0, Math.min(100, value));
},
```

3. Add selectors:
```js
export const selectFriendship = (npcId) => (state) => state.npc?.friendship?.[npcId] ?? 50;
export const selectFriendshipTier = (npcId) => (state) => {
  const val = state.npc?.friendship?.[npcId] ?? 50;
  if (val >= 75) return 'close';    // Unlocks special dialogue, gifts
  if (val >= 50) return 'friendly'; // Default neutral+
  if (val >= 25) return 'cautious'; // Some dialogue restricted
  return 'cold';                     // Minimal interaction
};
export const selectAllFriendships = (state) => state.npc?.friendship || {};
```

Create NEW file. A Redux middleware that listens to game actions and dispatches friendship adjustments:

```js
import { adjustFriendship } from './npcSlice.js';

// Friendship deltas by event type
const FRIENDSHIP_DELTAS = {
  correctAnswer: 2,      // Answered quiz correctly during NPC interaction
  gift: 5,               // Gave a gift to NPC
  questComplete: 10,     // Completed a quest for this NPC
  ignored: -1,           // Walked away during dialogue (future)
  wrongAnswer: -1,       // Wrong quiz answer during NPC interaction
};

export const friendshipMiddleware = (store) => (next) => (action) => {
  const result = next(action);

  // Listen to quiz completion during NPC context
  if (action.type === 'quiz/recordAnswer') {
    const { correct, context } = action.payload || {};
    if (context?.npcId) {
      const delta = correct ? FRIENDSHIP_DELTAS.correctAnswer : FRIENDSHIP_DELTAS.wrongAnswer;
      store.dispatch(adjustFriendship({ npcId: context.npcId, delta, reason: correct ? 'correctAnswer' : 'wrongAnswer' }));
    }
  }

  // Listen to quest completion
  if (action.type === 'quest/completeQuest') {
    const { questId, npcId } = action.payload || {};
    if (npcId) {
      store.dispatch(adjustFriendship({ npcId, delta: FRIENDSHIP_DELTAS.questComplete, reason: 'questComplete' }));
    }
  }

  // Listen to gift giving (from companion/NPC gift system)
  if (action.type === 'npc/giveGift' || action.type === 'companion/giveGift') {
    const { npcId } = action.payload || {};
    if (npcId) {
      store.dispatch(adjustFriendship({ npcId, delta: FRIENDSHIP_DELTAS.gift, reason: 'gift' }));
    }
  }

  return result;
};
```

NOTE: The exact action types (quiz/recordAnswer, quest/completeQuest) should be verified against the actual Redux slices. Read the quiz and quest slices to confirm the action type names. If they differ, adjust the strings accordingly.

This middleware is NOT wired to the store yet — it will be registered in a future step or can be added to store.js. For now, just create the file with the correct export.
