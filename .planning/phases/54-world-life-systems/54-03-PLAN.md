**2. Modify `src/components/World/ObjectInteractionOverlay.jsx`** — add FSRS dispatch for inscription words and root family word batch dispatch.

Add imports at the top:
```javascript
import { addFsrsCard } from '../../store/slices/vocabularySlice.js';
import { getRootWords } from '../../data/rootsData.js';
import { INSCRIPTION_DATA } from '../../data/inscriptionData.js';
import { store } from '../../store/store.js';
import TashkeelText from '../Arabic/TashkeelText.jsx';
```

Add a `useEffect` that fires when `data` changes and `data.type === 'inscription'`:

```javascript
import { useEffect } from 'react';

// Inside the component, after the useFocusTrap call:
useEffect(() => {
  if (!data || data.type !== 'inscription') return;

  const state = store.getState();
  const fsrsCards = state.vocabulary?.fsrsCards ?? {};

  // ENVR-03: Add unknown inscription words to FSRS queue
  const inscriptionContent = INSCRIPTION_DATA[data.id];
  const wordIds = inscriptionContent?.wordIds || data.rootWords || [];
  wordIds
    .filter(wordId => !fsrsCards[wordId])
    .forEach(wordId => {
      dispatch(addFsrsCard({
        wordId,
        card: {
          due: new Date().toISOString(),
          stability: 0, difficulty: 0, elapsed_days: 0,
          scheduled_days: 0, reps: 0, lapses: 0, state: 'New',
        },
        source: `inscription_${data.id}`,
      }));
    });

  // ENVR-04: Teach root family words if inscription has rootFamily
  if (data.rootFamily) {
    const rootWords = getRootWords(data.rootFamily) || [];
    rootWords
      .filter(wordId => !fsrsCards[wordId])
      .slice(0, 5)  // Cap at 5 to avoid overwhelming the queue
      .forEach(wordId => {
        dispatch(addFsrsCard({
          wordId,
          card: {
            due: new Date().toISOString(),
            stability: 0, difficulty: 0, elapsed_days: 0,
            scheduled_days: 0, reps: 0, lapses: 0, state: 'New',
          },
          source: `inscription_root_${data.rootFamily}`,
        }));
      });
  }
}, [data, dispatch]);
```
