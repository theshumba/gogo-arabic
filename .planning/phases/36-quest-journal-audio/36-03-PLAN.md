```js
import { store } from '../../store/store.js';
import { EventBus } from '../../utils/eventBus.js';
import { EVENTS } from '../../utils/eventBusTypes.js';

export class AutoSave {
  constructor(intervalMs = 180000) { // 3 minutes
    this.intervalMs = intervalMs;
    this.timer = null;
    this.saving = false;
  }

  start() {
    this.timer = setInterval(() => this._tryAutoSave(), this.intervalMs);
  }

  stop() {
    if (this.timer) clearInterval(this.timer);
    this.timer = null;
  }

  _tryAutoSave() {
    if (this.saving) return;
    if (!this._isSafeToSave()) return;

    this.saving = true;
    try {
      // Redux-persist handles the actual save via its persist middleware.
      // We just need to flush the persistor.
      // Emit an event so UI can show a brief save indicator.
      EventBus.emit(EVENTS.AUTOSAVE_TRIGGERED);

      // If using redux-persist, call persistor.flush()
      // The persistor is typically exported from store.js
      // For now, emit the event — the save happens automatically via redux-persist
    } finally {
      this.saving = false;
    }
  }

  _isSafeToSave() {
    const state = store.getState();

    // Check various unsafe states
    const ui = state.ui || {};
    if (ui.showBattle || ui.battleActive) return false;
    if (ui.showDialogue || ui.dialogueActive) return false;
    if (ui.showQuiz || ui.quizActive) return false;
    if (ui.isTransitioning) return false;

    // Check for any active overlay that might cause state inconsistency
    if (ui.frozen) return false;

    return true;
  }
}
```
