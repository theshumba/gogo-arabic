    ### 1. Create src/game/systems/FloatingArabicLabelManager.js

    ```js
    import { store } from '../../store/store.js';
    import { getRetrievability } from '../../services/fsrs.js';
    import { getLabelsForZone } from '../../data/environmentalLabels.js';
    import { createArabicText } from '../ui/ArabicText.js';

    /**
     * FloatingArabicLabelManager
     *
     * Renders floating Arabic vocabulary labels above world objects.
     * Labels fade based on FSRS mastery:
     * - stability < 7 days: full opacity (1.0) with tashkeel
     * - stability 7-30 days: partial opacity (0.5), partial tashkeel
     * - stability > 30 days: hidden (word mastered — passive learning complete)
     * - Unknown word (no FSRS card): full opacity + transliteration shown
     *
     * IMM-03: Environmental Arabic labels teach vocabulary passively.
     */

    const LABEL_OFFSET_Y = -65;       // Above the InteractableManager label at -50
    const TRANSLIT_OFFSET_Y = -50;    // Below Arabic, at same level as interactable label
    const UPDATE_INTERVAL = 5000;     // Re-check FSRS every 5 seconds (not every frame)

    export class FloatingArabicLabelManager {
      constructor(scene) {
        this.scene = scene;
        this.labels = [];              // Array of { labelDef, arabicText, translitText, lastAlpha }
        this._lastUpdateTime = 0;
      }

      /**
       * Create floating labels for the current zone.
       * Must be called AFTER interactableManager.create() so we can find sprite positions.
       *
       * @param {string} zoneName - Current zone ID
       * @param {InteractableManager} interactableManager - To look up object world positions
       */
      create(zoneName, interactableManager) {
        this.destroy(); // Clean up any existing labels

        const labelDefs = getLabelsForZone(zoneName);
        if (!labelDefs.length) return;

        const interactables = interactableManager.getInteractables?.() || [];

        for (const def of labelDefs) {
          // Find the matching interactable by ID
          const obj = interactables.find(i => i.id === def.interactableId);
          if (!obj) continue; // Object not found in current zone — skip

          const px = obj.worldX;
          const py = obj.worldY;

          // Create Arabic text label above object (above the existing interactable label)
          const arabicText = createArabicText(
            this.scene, px, py + LABEL_OFFSET_Y, def.arabic,
            { fontSize: '16px', color: '#e2b659', stroke: '#1a1a2e', strokeThickness: 4 }
          );
          arabicText.setOrigin(0.5).setDepth(9998);

          // Create transliteration text (shown for unknown/new words)
          const translitText = this.scene.add.text(px, py + TRANSLIT_OFFSET_Y, def.transliteration, {
            fontFamily: "'Press Start 2P', monospace",
            fontSize: '6px',
            color: '#c8c8c8',
            stroke: '#1a1a2e',
            strokeThickness: 2,
          }).setOrigin(0.5).setDepth(9998);

          // Add gentle bob animation
          this.scene.tweens.add({
            targets: [arabicText],
            y: py + LABEL_OFFSET_Y - 4,
            duration: 2000,
            yoyo: true,
            repeat: -1,
            ease: 'Sine.easeInOut',
          });

          this.labels.push({
            def,
            arabicText,
            translitText,
            worldX: px,
            worldY: py,
            lastAlpha: 1,
          });
        }

        // Initial visibility update
        this._updateVisibility();
      }

      /**
       * Update label visibility based on FSRS mastery.
       * Called periodically (every UPDATE_INTERVAL ms), not every frame.
       */
      update(time) {
        if (time - this._lastUpdateTime < UPDATE_INTERVAL) return;
        this._lastUpdateTime = time;
        this._updateVisibility();
      }

      _updateVisibility() {
        const state = store.getState();
        const fsrsCards = state.vocabulary?.fsrsCards ?? {};
        const now = new Date();

        for (const label of this.labels) {
          const cardData = fsrsCards[label.def.vocabWordId];

          if (!cardData || !cardData.card || cardData.card.reps === 0) {
            // Unknown word — full visibility + transliteration
            label.arabicText.setAlpha(1.0);
            label.translitText.setVisible(true);
            label.translitText.setAlpha(0.8);
            continue;
          }

          const card = cardData.card;
          const stability = card.stability || 0;

          if (stability < 7) {
            // Fresh — full tashkeel, full opacity, show transliteration
            label.arabicText.setAlpha(1.0);
            label.translitText.setVisible(true);
            label.translitText.setAlpha(0.6);
          } else if (stability < 30) {
            // Partial mastery — reduced opacity, hide transliteration
            label.arabicText.setAlpha(0.5);
            label.translitText.setVisible(false);
          } else {
            // Mastered — hide label entirely
            label.arabicText.setAlpha(0);
            label.translitText.setVisible(false);
          }
        }
      }

      /**
       * Clean up all label objects.
       * Called in WorldScene.clearZone() before loading a new zone.
       */
      destroy() {
        for (const label of this.labels) {
          if (label.arabicText?.destroy) label.arabicText.destroy();
          if (label.translitText?.destroy) label.translitText.destroy();
        }
        this.labels = [];
        this._lastUpdateTime = 0;
      }
    }
    ```
