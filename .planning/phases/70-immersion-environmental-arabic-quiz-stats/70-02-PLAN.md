    ### 1. Create src/services/quizStatAccumulator.js

    A pure localStorage service with no Redux dependency.

    ```js
    /**
     * QuizStatAccumulator
     *
     * Tracks per-question correctness across play sessions using localStorage.
     * Stats are keyed by `wordId::quizType` for granular tracking.
     *
     * IMM-04: Anonymous quiz statistics — "72% of players get this right"
     *
     * Data shape in localStorage:
     * {
     *   sessionCount: number,           // total sessions that have recorded stats
     *   currentSessionId: string,       // ISO date + random suffix to identify session
     *   stats: {
     *     [key: string]: {              // key = `${wordId}::${quizType}`
     *       correct: number,
     *       total: number,
     *       sessions: number,           // number of unique sessions that saw this question
     *     }
     *   }
     * }
     */

    const STORAGE_KEY = 'gogo_quiz_stats';
    const MIN_SESSIONS_FOR_DISPLAY = 3; // Minimum sessions before showing stats

    /**
     * Get or initialize the stats object from localStorage.
     */
    function _getStore() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) return JSON.parse(raw);
      } catch {
        // Corrupted data — reset
      }
      return { sessionCount: 0, currentSessionId: null, stats: {} };
    }

    /**
     * Persist the stats object to localStorage.
     */
    function _setStore(data) {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
      } catch {
        // localStorage full or blocked — fail silently
      }
    }

    /**
     * Get or create a session ID for the current play session.
     * A new session is started if the stored session ID is null or
     * from a different calendar day + random suffix.
     */
    function _ensureSession(data) {
      if (!data.currentSessionId) {
        data.currentSessionId = `${new Date().toISOString().split('T')[0]}_${Math.random().toString(36).slice(2, 6)}`;
        data.sessionCount += 1;
      }
      return data;
    }

    /**
     * Build the storage key for a question.
     */
    function _key(wordId, quizType) {
      return `${wordId}::${quizType}`;
    }

    /**
     * Record an answer for a quiz question.
     * Called from useQuiz.answer() after each response.
     *
     * @param {string} wordId - Vocabulary word ID
     * @param {string} quizType - Quiz type (e.g. 'ar-to-en', 'GrammarFill')
     * @param {boolean} correct - Whether the answer was correct
     */
    export function recordAnswer(wordId, quizType, correct) {
      if (!wordId || !quizType) return;

      const data = _getStore();
      _ensureSession(data);

      const k = _key(wordId, quizType);
      if (!data.stats[k]) {
        data.stats[k] = { correct: 0, total: 0, sessions: 0 };
      }

      const stat = data.stats[k];
      stat.total += 1;
      if (correct) stat.correct += 1;

      // Track unique sessions per question (approximate — increment once per session)
      // Use a Set stored as array to track session IDs per question
      // For simplicity, we track sessions as a count and accept slight over-counting
      // A more precise approach would store session IDs, but that bloats localStorage
      if (!stat._lastSession || stat._lastSession !== data.currentSessionId) {
        stat.sessions += 1;
        stat._lastSession = data.currentSessionId;
      }

      _setStore(data);
    }

    /**
     * Get statistics for a specific question.
     *
     * @param {string} wordId - Vocabulary word ID
     * @param {string} quizType - Quiz type
     * @returns {{ percent: number, total: number, sessions: number } | null}
     *   Returns null if insufficient data (fewer than MIN_SESSIONS_FOR_DISPLAY sessions).
     */
    export function getStats(wordId, quizType) {
      if (!wordId || !quizType) return null;

      const data = _getStore();
      const k = _key(wordId, quizType);
      const stat = data.stats[k];

      if (!stat || stat.sessions < MIN_SESSIONS_FOR_DISPLAY || stat.total < 3) {
        return null; // Not enough data to display meaningful stats
      }

      return {
        percent: Math.round((stat.correct / stat.total) * 100),
        total: stat.total,
        sessions: stat.sessions,
      };
    }

    /**
     * End the current session (call on quiz overlay close or app unmount).
     * Clears currentSessionId so the next recordAnswer starts a new session.
     */
    export function endSession() {
      const data = _getStore();
      data.currentSessionId = null;
      _setStore(data);
    }

    /**
     * Get the total number of questions with valid stats (for progress tracking).
     * @returns {number}
     */
    export function getStatsCount() {
      const data = _getStore();
      return Object.values(data.stats).filter(
        s => s.sessions >= MIN_SESSIONS_FOR_DISPLAY && s.total >= 3
      ).length;
    }

    /**
     * Reset all stats (for testing or settings).
     */
    export function resetStats() {
      localStorage.removeItem(STORAGE_KEY);
    }
    ```
