    ```js
    // src/dev/hmrContent.js
    // Phase 104 HMR boundary — DEV ONLY. Entire module guarded by `if (import.meta.hot)`
    // so Rollup tree-shakes it from production builds (per Vite HMR docs).

    if (import.meta.hot) {
      const { store } = await import('../store/index.js');
      const { hotSwap } = await import('../store/slices/contentSlice.js');
      const { ContentSchema } = await import('../data/contentSchema.js');
      const { showHmrError, clearHmrError } = await import('../utils/devOverlay.js');

      // CRITICAL: eager:true required — without it Vite doesn't register HMR deps
      // (RESEARCH Pitfall 2 — github.com/vitejs/vite discussion #7577)
      const modules = import.meta.glob('/content/**/*.json', { eager: true });

      // Initial dispatch on dev-server start so the store has the current content
      const initial = {};
      for (const [filePath, mod] of Object.entries(modules)) {
        initial[filePath] = mod?.default ?? mod;
      }
      store.dispatch(hotSwap(initial));

      import.meta.hot.accept(Object.keys(modules), (newModules) => {
        const composed = {};
        const keys = Object.keys(modules);
        newModules.forEach((mod, i) => {
          if (mod) composed[keys[i]] = mod?.default ?? mod;
        });

        const result = ContentSchema.safeParse(composed);
        if (!result.success) {
          const issues = result.error.issues.map(i =>
            `[${i.path.join('.')}] ${i.message}`
          );
          // Report the first file path mentioned in the issues for context
          const firstFile = result.error.issues[0]?.path?.[0] ?? 'content/**/*.json';
          showHmrError({ file: String(firstFile), issues });
          return;  // state NOT mutated
        }
        clearHmrError();
        store.dispatch(hotSwap(composed));
      });
    }
    ```
