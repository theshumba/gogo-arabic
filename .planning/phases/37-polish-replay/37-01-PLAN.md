Create NEW file. A utility that shuffles which vocabulary words are assigned to which zones, enabling different playthroughs.

Key design:
- Takes the full vocabulary dataset and zone list as inputs
- Uses a seeded random (seed from settingsSlice or player save) for reproducible shuffles
- Constraints: words in the same category (greetings, food, etc.) stay together — don't scatter a category across 8 zones
- Constraints: each zone gets roughly equal word count
- Constraints: beginner words (A1 CEFR) always assigned to oasis_village (starter zone)

```js
export class VocabRandomizer {
  constructor(seed = Date.now()) {
    this.seed = seed;
  }

  // Seeded PRNG (mulberry32)
  _random() {
    let t = this.seed += 0x6D2B79F5;
    t = Math.imul(t ^ t >>> 15, t | 1);
    t ^= t + Math.imul(t ^ t >>> 7, t | 61);
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  }

  // Fisher-Yates shuffle using seeded random
  _shuffle(arr) {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(this._random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  }

  randomize(vocabulary, zones) {
    // Group words by category
    const categories = {};
    for (const word of vocabulary) {
      const cat = word.category || 'general';
      if (!categories[cat]) categories[cat] = [];
      categories[cat].push(word);
    }

    // Shuffle category order
    const catNames = this._shuffle(Object.keys(categories));

    // Assign categories to zones round-robin
    // Keep A1/beginner categories in first zone (oasis_village)
    const zoneAssignments = {};
    for (const z of zones) zoneAssignments[z] = [];

    const zoneList = [...zones];
    let zoneIdx = 0;

    for (const catName of catNames) {
      const words = categories[catName];
      // A1 words go to first zone
      const isBeginnerCat = words.every(w => w.cefr === 'A1');
      const targetZone = isBeginnerCat ? zoneList[0] : zoneList[zoneIdx % zoneList.length];
      zoneAssignments[targetZone].push(...words);
      if (!isBeginnerCat) zoneIdx++;
    }

    return zoneAssignments; // { zoneName: [word, word, ...] }
  }
}
```
